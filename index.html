async function settleUp() { if (transactions.length === 0) return alert('未精算の記録がありません。'); if (confirm('現在の精算を完了し、残高を0にリセットしますか？')) { const settledDate = new Date().toLocaleString('ja-JP'); // --- ここから追加: 精算時点での最終精算額を計算 --- let balance = 0; transactions.forEach((t) => { let yRatio = 1, uRatio = 1; if (t.splitType === 'CUSTOM') { yRatio = t.ratioY !== undefined ? t.ratioY : 1; uRatio = t.ratioU !== undefined ? t.ratioU : 1; } else if (t.splitType === 'PAY_ALL') { if (t.payer === 'ゆうすけ') { yRatio = 0; uRatio = 1; } else { yRatio = 1; uRatio = 0; } } else if (t.splitType && t.splitType.startsWith('RATIO_')) { const parts = t.splitType.split('_'); yRatio = parseInt(parts[1], 10); uRatio = parseInt(parts[2], 10); } const totalRatio = yRatio + uRatio; if (totalRatio > 0) { if (t.payer === 'ゆうすけ') { balance += Math.round((t.amount * uRatio) / totalRatio); } else if (t.payer === 'うらら') { balance -= Math.round((t.amount * yRatio) / totalRatio); } } }); let summaryText = "貸し借りなし（清算済み）"; if (balance > 0) { summaryText = `うらら ➔ ゆうすけ へ ${balance.toLocaleString()}円`; } else if (balance < 0) { summaryText = `ゆうすけ ➔ うらら へ ${Math.abs(balance).toLocaleString()}円`; } // --- ここまで --- const itemsToSave = [...transactions].sort((a, b) => a.createdAt - b.createdAt); await db.collection('settledGroups').add({ date: settledDate, items: itemsToSave, summaryText: summaryText, // ← 清算結果テキストを保存 createdAt: Date.now() }); const batch = db.batch(); transactions.forEach(t => { const ref = db.collection('transactions').doc(t.id); batch.delete(ref); }); await batch.commit(); } }
